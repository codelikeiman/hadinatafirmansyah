// ─── UTILS ────────────────────────────────────────────────────────────────────
function useReveal(threshold = 0.12) {
  const ref = useRef(null);
  const [vis, setVis] = useState(false);
  useEffect(() => {
    const obs = new IntersectionObserver(([e]) => { if (e.isIntersecting) { setVis(true); obs.disconnect(); } }, { threshold });
    if (ref.current) obs.observe(ref.current);
    return () => obs.disconnect();
  }, []);
  return [ref, vis];
}

function useSection(ids) {
  const [active, setActive] = useState(ids[0]);
  useEffect(() => {
    const obs = new IntersectionObserver((entries) => {
      for (const e of entries) { if (e.isIntersecting) setActive(e.target.id); }
    }, { threshold: 0.3 });
    ids.forEach(id => { const el = document.getElementById(id); if (el) obs.observe(el); });
    return () => obs.disconnect();
  }, []);
  return active;
}

// ─── CANVAS PARTICLES ─────────────────────────────────────────────────────────
function ParticleCanvas() {
  const ref = useRef(null);
  const mouse = useRef({ x: -999, y: -999 });
  useEffect(() => {
    const c = ref.current; if (!c) return;
    const ctx = c.getContext("2d");
    let W, H, raf;
    const resize = () => { W = c.width = c.offsetWidth; H = c.height = c.offsetHeight; };
    resize();
    const pts = Array.from({ length: 55 }, () => ({
      x: Math.random() * W, y: Math.random() * H,
      vx: (Math.random() - .5) * .28, vy: (Math.random() - .5) * .28,
      s: Math.random() * 1.2 + .4,
    }));
    const onMM = e => { mouse.current = { x: e.clientX, y: e.clientY }; };
    window.addEventListener("resize", resize);
    window.addEventListener("mousemove", onMM);
    const draw = () => {
      ctx.clearRect(0, 0, W, H);
      for (const p of pts) {
        const dx = p.x - mouse.current.x, dy = p.y - mouse.current.y;
        const d = Math.hypot(dx, dy);
        if (d < 110) { const f = (110 - d) / 110 * .45; p.vx += dx / d * f; p.vy += dy / d * f; }
        p.vx *= .97; p.vy *= .97;
        p.x += p.vx; p.y += p.vy;
        if (p.x < 0) { p.x = 0; p.vx *= -1; } if (p.x > W) { p.x = W; p.vx *= -1; }
        if (p.y < 0) { p.y = 0; p.vy *= -1; } if (p.y > H) { p.y = H; p.vy *= -1; }
      }
      for (let i = 0; i < pts.length; i++) {
        for (let j = i + 1; j < pts.length; j++) {
          const dist = Math.hypot(pts[i].x - pts[j].x, pts[i].y - pts[j].y);
          if (dist < 120) {
            ctx.beginPath();
            ctx.strokeStyle = `rgba(30,90,130,${(1 - dist / 120) * .22})`;
            ctx.lineWidth = .5;
            ctx.moveTo(pts[i].x, pts[i].y);
            ctx.lineTo(pts[j].x, pts[j].y);
            ctx.stroke();
          }
        }
        ctx.beginPath();
        ctx.arc(pts[i].x, pts[i].y, pts[i].s, 0, Math.PI * 2);
        ctx.fillStyle = "rgba(0,240,160,.35)";
        ctx.fill();
      }
      raf = requestAnimationFrame(draw);
    };
    raf = requestAnimationFrame(draw);
    return () => { window.removeEventListener("resize", resize); window.removeEventListener("mousemove", onMM); cancelAnimationFrame(raf); };
  }, []);
  return <canvas ref={ref} style={{ position: "fixed", inset: 0, width: "100%", height: "100%", zIndex: 0, pointerEvents: "none" }} />;
}

// ─── CUSTOM CURSOR ────────────────────────────────────────────────────────────
function Cursor() {
  const dot = useRef(null), ring = useRef(null);
  const pos = useRef({ x: -100, y: -100 }), lag = useRef({ x: -100, y: -100 });
  useEffect(() => {
    const onMM = e => { pos.current = { x: e.clientX, y: e.clientY }; };
    const onOver = e => { if (e.target.closest("button,a,input,textarea")) { dot.current?.classList.add("big"); ring.current?.classList.add("big"); } };
    const onOut = () => { dot.current?.classList.remove("big"); ring.current?.classList.remove("big"); };
    document.addEventListener("mousemove", onMM);
    document.addEventListener("mouseover", onOver);
    document.addEventListener("mouseout", onOut);
    let r;
    const tick = () => {
      lag.current.x += (pos.current.x - lag.current.x) * .1;
      lag.current.y += (pos.current.y - lag.current.y) * .1;
      if (dot.current) dot.current.style.cssText = `left:${pos.current.x}px;top:${pos.current.y}px`;
      if (ring.current) ring.current.style.cssText = `left:${lag.current.x}px;top:${lag.current.y}px`;
      r = requestAnimationFrame(tick);
    };
    r = requestAnimationFrame(tick);
    return () => { document.removeEventListener("mousemove", onMM); document.removeEventListener("mouseover", onOver); document.removeEventListener("mouseout", onOut); cancelAnimationFrame(r); };
  }, []);
  return (<><div ref={dot} className="cdot" /><div ref={ring} className="cring" /></>);
}

// ─── SCROLL PROGRESS ─────────────────────────────────────────────────────────
function ScrollProgress() {
  const [pct, setPct] = useState(0);
  useEffect(() => {
    const fn = () => { const t = document.documentElement.scrollHeight - window.innerHeight; setPct(t > 0 ? (window.scrollY / t) * 100 : 0); };
    window.addEventListener("scroll", fn);
    return () => window.removeEventListener("scroll", fn);
  }, []);
  return <div className="progress-bar"><div className="progress-fill" style={{ width: `${pct}%` }} /></div>;
}

// ─── TECH GRAPH (D3) ──────────────────────────────────────────────────────────
function TechGraph() {
  const svgRef = useRef(null);
  const [ref, vis] = useReveal();
  const [hovered, setHovered] = useState(null);
  const started = useRef(false);

  useEffect(() => {
    if (!vis || started.current || !svgRef.current) return;
    started.current = true;

    const el = svgRef.current;
    const svg = d3.select(el);
    const W = el.clientWidth || 680;
    const H = el.clientHeight || 460;

    const nodes = GRAPH_NODES.map(d => ({ ...d }));
    const links = GRAPH_LINKS.map(([s, t]) => ({ source: s, target: t }));

    const sim = d3.forceSimulation(nodes)
      .force("link", d3.forceLink(links).id(d => d.id).distance(72).strength(.6))
      .force("charge", d3.forceManyBody().strength(-220))
      .force("center", d3.forceCenter(W / 2, H / 2))
      .force("collide", d3.forceCollide(28));

    const linkSel = svg.append("g").selectAll("line").data(links).enter().append("line")
      .attr("stroke", "rgba(30,70,100,.45)").attr("stroke-width", 1);

    const nodeSel = svg.append("g").selectAll("g").data(nodes).enter().append("g")
      .call(d3.drag()
        .on("start", (ev, d) => { if (!ev.active) sim.alphaTarget(.3).restart(); d.fx = d.x; d.fy = d.y; })
        .on("drag", (ev, d) => { d.fx = ev.x; d.fy = ev.y; })
        .on("end", (ev, d) => { if (!ev.active) sim.alphaTarget(0); d.fx = null; d.fy = null; }));

    nodeSel.append("circle")
      .attr("r", d => 6 + d.lv * 0.1)
      .attr("fill", d => G_COLORS[d.g])
      .attr("opacity", .8)
      .attr("stroke", d => G_COLORS[d.g])
      .attr("stroke-width", .5)
      .attr("stroke-opacity", .3);

    nodeSel.append("circle")
      .attr("r", d => 10 + d.lv * 0.1)
      .attr("fill", "transparent")
      .attr("stroke", d => G_COLORS[d.g])
      .attr("stroke-width", .5)
      .attr("stroke-opacity", .15);

    nodeSel.append("text")
      .text(d => d.id)
      .attr("dy", d => -(8 + d.lv * 0.1 + 4))
      .attr("text-anchor", "middle")
      .attr("fill", "#7fa8c0")
      .attr("font-size", "9px")
      .attr("font-family", "JetBrains Mono, monospace")
      .attr("letter-spacing", "1px")
      .attr("pointer-events", "none");

    sim.on("tick", () => {
      linkSel.attr("x1", d => d.source.x).attr("y1", d => d.source.y)
        .attr("x2", d => d.target.x).attr("y2", d => d.target.y);
      nodeSel.attr("transform", d => `translate(${d.x},${d.y})`);
    });

    return () => { sim.stop(); svg.selectAll("*").remove(); };
  }, [vis]);

  const legendGroups = [["runtime", "Runtime"], ["lang", "Language"], ["fw", "Framework"], ["db", "Database"], ["cache", "Cache"], ["infra", "Infra"], ["api", "API"], ["auth", "Auth"], ["mq", "Messaging"], ["test", "Testing"], ["cloud", "Cloud"]];

  return (
    <section id="graph" className="sec">
      <div className="sec-label">// 04 — DEPENDENCY GRAPH</div>
      <h2 className="sec-title">TECH <span>ECOSYSTEM</span></h2>

      <div ref={ref} className={`reveal${vis ? " show" : ""} sec-win`}>
        <div className="win-bar">
          <div className="win-buttons">{["#ff5f56", "#ffbd2e", "#27c93f"].map((c, i) => <div key={i} className="win-btn" style={{ background: c }} />)}</div>
          <span className="win-title">graph --layout=force --directed=false</span>
          <div className="win-tag"><span style={{ color: "var(--tx2)" }}>DRAG</span> nodes to rearrange</div>
        </div>
        <div className="graph-container">
          <svg ref={svgRef} />
          <div className="graph-hint">D3 FORCE SIMULATION · {GRAPH_NODES.length} NODES · {GRAPH_LINKS.length} EDGES</div>
        </div>
      </div>

      <div className="graph-legend" style={{ marginTop: 14 }}>
        {legendGroups.map(([g, label]) => (
          <div key={g} className="legend-item">
            <div className="legend-dot" style={{ background: G_COLORS[g] }} />
            <span>{label}</span>
          </div>
        ))}
      </div>
    </section>
  );
}

  const [lines, setLines] = useState([{ type: "welcome", text: 'IMAN.OS v2.4.1-lts · Type "help" for available commands.' }]);
  const [input, setInput] = useState("");
  const [hist, setHist] = useState([]);
  const [hi, setHi] = useState(-1);
  const bodyRef = useRef(null);
  const inpRef = useRef(null);

  useEffect(() => { if (bodyRef.current) bodyRef.current.scrollTop = bodyRef.current.scrollHeight; }, [lines]);

  const parseAnsi = useCallback((text) => {
    const parts = [];
    const re = /\x1b\[([a-z]+)\](.*?)\x1b\[\/\1\]/g;
    let last = 0, m;
    while ((m = re.exec(text)) !== null) {
      if (m.index > last) parts.push({ t: text.slice(last, m.index), c: null });
      const colorMap = { g: "var(--g)", c: "var(--c)", a: "var(--a)", r: "var(--r)", b: "var(--b)", d: "var(--tx3)" };
      parts.push({ t: m[2], c: colorMap[m[1]] || (m[1] === "b" ? null : "var(--bright)"), bold: m[1] === "b" });
      last = m.index + m[0].length;
    }
    if (last < text.length) parts.push({ t: text.slice(last), c: null });
    return parts;
  }, []);

  const run = useCallback((cmd) => {
    const c = cmd.trim().toLowerCase();
    setHist(h => [...h, cmd]); setHi(-1);
    if (c === "clear") { setLines([]); return; }
    const fn = TERM_COMMANDS[c];
    const out = fn ? fn() : `bash: ${cmd}: command not found\nType "help" to see available commands.`;
    setLines(l => [...l, { type: "cmd", text: cmd }, { type: "out", text: out, isErr: !fn }]);
  }, []);

  const onKey = useCallback((e) => {
    if (e.key === "Enter") { if (input.trim()) run(input); setInput(""); }
    else if (e.key === "ArrowUp") { e.preventDefault(); const ni = Math.min(hi + 1, hist.length - 1); setHi(ni); setInput(hist[hist.length - 1 - ni] || ""); }
    else if (e.key === "ArrowDown") { e.preventDefault(); const ni = Math.max(hi - 1, -1); setHi(ni); setInput(ni === -1 ? "" : hist[hist.length - 1 - ni] || ""); }
    else if (e.key === "Tab") { e.preventDefault(); const m = Object.keys(TERM_COMMANDS).find(k => k.startsWith(input)); if (m) setInput(m); }
  }, [input, hist, hi, run]);

  const renderLine = (line, i) => {
    if (line.type === "welcome") return <div key={i} className="term-out" style={{ color: "var(--c)" }}>{line.text}</div>;
    if (line.type === "cmd") return (
      <div key={i} className="term-out">
        <span style={{ color: "var(--g)" }}>iman@backend.dev</span>
        <span style={{ color: "var(--tx3)" }}>:</span>
        <span style={{ color: "var(--b)" }}>~</span>
        <span style={{ color: "var(--tx)" }}>$ </span>
        <span style={{ color: "var(--bright)" }}>{line.text}</span>
      </div>
    );
    const parts = parseAnsi(line.text);
    return (
      <div key={i} className="term-out" style={{ color: line.isErr ? "var(--r)" : "var(--tx2)" }}>
        <pre style={{ margin: 0, fontFamily: "var(--fm)", whiteSpace: "pre-wrap", wordBreak: "break-word" }}>
          {parts.map((p, j) => <span key={j} style={{ color: p.c || undefined, fontWeight: p.bold ? 700 : undefined }}>{p.t}</span>)}
        </pre>
      </div>
    );
  };

  return (
    <div className="terminal" onClick={() => inpRef.current?.focus()}>
      <div className="term-bar">
        <div className="term-btns">{["#ff5f56", "#ffbd2e", "#27c93f"].map((c, i) => <div key={i} className="term-btn" style={{ background: c }} />)}</div>
        <span className="term-title">iman@backend.dev — bash</span>
        <span style={{ fontFamily: "var(--fm)", fontSize: 8, color: "var(--tx3)", letterSpacing: 2 }}>5.2.15</span>
      </div>
      <div className="term-body" ref={bodyRef}>
        {lines.map((l, i) => renderLine(l, i))}
        <div className="term-irow">
          <span className="term-prompt">
            <span style={{ color: "var(--g)" }}>iman@backend.dev</span>
            <span style={{ color: "var(--tx3)" }}>:</span>
            <span style={{ color: "var(--b)" }}>~</span>
            <span style={{ color: "var(--tx)" }}>$ </span>
          </span>
          <input ref={inpRef} className="term-inp" value={input} onChange={e => setInput(e.target.value)} onKeyDown={onKey} autoComplete="off" spellCheck={false} autoFocus />
        </div>
      </div>
    </div>
  );
}

function useReveal() {
  const ref = useRef(null);
  const [visible, setVisible] = useState(false);

  useEffect(() => {
    const node = ref.current;
    if (!node) return;
    const obs = new IntersectionObserver(
      ([entry]) => {
        if (entry.isIntersecting) setVisible(true);
      },
      { threshold: 0.18 }
    );
    obs.observe(node);
    return () => obs.disconnect();
  }, []);

  return [ref, visible];
}

function ASCIIHeroArt({ data }) {
  const rows = data?.rows ?? [];
  const columns = data?.columns ?? rows[0]?.text?.length ?? 0;

  const scale = useMemo(() => {
    if (!columns) return 1;
    if (columns >= 140) return 0.66;
    if (columns >= 120) return 0.78;
    return 1;
  }, [columns]);

  return (
    <div className="ascii-stage">
      <div className="ascii-scaler" style={{ transform: `scale(${scale})` }}>
        <div className="ascii-grid" aria-label="ASCII art portrait">
          {rows.map((row, rowIndex) => (
            <div className="ascii-row" key={rowIndex}>
              {Array.from(row.text || "").map((char, colIndex) => {
                const color = row.colors?.[colIndex] ?? undefined;
                const empty = char === " ";
                return (
                  <span
                    key={`${rowIndex}-${colIndex}`}
                    className={`ascii-char${empty ? " ascii-empty" : ""}`}
                    style={{ color }}
                  >
                    {char}
                  </span>
                );
              })}
            </div>
          ))}
        </div>
      </div>
    </div>
  );
}

function GraphSection() {
  const [ref, vis] = useReveal();
  const svgRef = useRef(null);

  useEffect(() => {
    if (!vis || !svgRef.current) return;

    const el = svgRef.current;
    const width = el.clientWidth || 900;
    const height = el.clientHeight || 480;

    d3.select(el).selectAll("*").remove();

    const svg = d3.select(el).append("svg").attr("width", width).attr("height", height);

    const nodes = GRAPH_NODES.map((d) => ({ ...d }));
    const links = GRAPH_LINKS.map(([source, target]) => ({ source, target }));

    const simulation = d3.forceSimulation(nodes)
      .force("link", d3.forceLink(links).id((d) => d.id).distance(90))
      .force("charge", d3.forceManyBody().strength(-240))
      .force("center", d3.forceCenter(width / 2, height / 2));

    const link = svg.append("g")
      .attr("stroke", "rgba(183,206,162,.18)")
      .selectAll("line")
      .data(links)
      .join("line")
      .attr("stroke-width", 1.2);

    const node = svg.append("g")
      .selectAll("circle")
      .data(nodes)
      .join("circle")
      .attr("r", (d) => Math.max(8, d.lv / 6))
      .attr("fill", (d) => G_COLORS[d.g] || "#b7cea2")
      .attr("stroke", "#08100b")
      .attr("stroke-width", 2);

    const label = svg.append("g")
      .selectAll("text")
      .data(nodes)
      .join("text")
      .text((d) => d.id)
      .attr("fill", "#dfeaed")
      .attr("font-size", 11)
      .attr("font-family", "JetBrains Mono");

    simulation.on("tick", () => {
      link
        .attr("x1", (d) => d.source.x)
        .attr("y1", (d) => d.source.y)
        .attr("x2", (d) => d.target.x)
        .attr("y2", (d) => d.target.y);

      node
        .attr("cx", (d) => d.x)
        .attr("cy", (d) => d.y);

      label
        .attr("x", (d) => d.x + 12)
        .attr("y", (d) => d.y + 4);
    });

    return () => simulation.stop();
  }, [vis]);

  return (
    <section id="graph" className="sec">
      <div className="sec-label">// 04 — relation graph</div>
      <h2 className="sec-title">Skill <span>Network</span></h2>

      <div ref={ref} className={`reveal ${vis ? "show" : ""} sec-win`}>
        <WindowBar title="graph.skill-map --engine=d3" tag="rendered" />
        <div className="graph-wrap">
          <div ref={svgRef} className="graph-box" />
        </div>
      </div>
    </section>
  );
}

function TerminalSection() {
  const [ref, vis] = useReveal();
  const [lines, setLines] = useState([
    "Terminal initialized.",
    "Type 'help' to list available commands.",
  ]);
  const [value, setValue] = useState("");
  const screenRef = useRef(null);

  useEffect(() => {
    if (screenRef.current) {
      screenRef.current.scrollTop = screenRef.current.scrollHeight;
    }
  }, [lines]);

  const runCommand = useCallback((raw) => {
    const cmd = raw.trim();
    if (!cmd) return;

    setLines((prev) => [...prev, `$ ${cmd}`]);

    const action = TERM_COMMANDS[cmd];
    if (!action) {
      setLines((prev) => [...prev, `Command not found: ${cmd}`]);
      return;
    }

    const result = action();
    if (result === "__CLEAR__") {
      setLines([]);
    } else {
      setLines((prev) => [...prev, result]);
    }
  }, []);

  return (
    <section id="terminal" className="sec">
      <div className="sec-label">// 05 — shell</div>
      <h2 className="sec-title">Interactive <span>Terminal</span></h2>

      <div ref={ref} className={`reveal ${vis ? "show" : ""} sec-win`}>
        <WindowBar title="portfolio.shell" tag="interactive" />

        <div className="terminal">
          <div ref={screenRef} className="term-screen">
            {lines.map((line, idx) => (
              <div key={idx} className="term-line">{line}</div>
            ))}
          </div>

          <div className="term-input-row">
            <span className="term-prompt">$</span>
            <input
              className="term-input"
              value={value}
              onChange={(e) => setValue(e.target.value)}
              onKeyDown={(e) => {
                if (e.key === "Enter") {
                  runCommand(value);
                  setValue("");
                }
              }}
              placeholder="type help"
            />
          </div>
        </div>
      </div>
    </section>
  );
}

export default function PortfolioMerged() {
  const [progress, setProgress] = useState(0);
  const [active, setActive] = useState("home");

  useEffect(() => {
    const onScroll = () => {
      const scrollTop = window.scrollY;
      const docHeight = document.documentElement.scrollHeight - window.innerHeight;
      const pct = docHeight > 0 ? (scrollTop / docHeight) * 100 : 0;
      setProgress(pct);

      const ids = ["home", "about", "skills", "projects", "graph", "terminal"];
      let current = "home";
      for (const id of ids) {
        const el = document.getElementById(id);
        if (!el) continue;
        const top = el.getBoundingClientRect().top;
        if (top <= 180) current = id;
      }
      setActive(current);
    };

    onScroll();
    window.addEventListener("scroll", onScroll, { passive: true });
    return () => window.removeEventListener("scroll", onScroll);
  }, []);

  return (
    <>
      <style>{CSS}</style>

      <div className="progress-bar">
        <div className="progress-fill" style={{ width: `${progress}%` }} />
      </div>

      <header className="statusbar">
        <div className="sb-left">
          <span className="sb-dot" />
          <span className="sb-brand">Julian Caesar Eben Ezer Simorangkir</span>
        </div>

        <div className="sb-right">
          <span><strong>Mode</strong> Portfolio</span>
          <span><strong>Theme</strong> ASCII Sync</span>
        </div>
      </header>

      <SidePanel />
      <NavDots active={active} />

      <main className="main">
        <Hero />
        <About />
        <Skills />
        <Projects />
        <GraphSection />
        <TerminalSection />
      </main>

      <footer className="footer">
        Portfolio system online · hero merged with custom ASCII art
      </footer>
    </>
  );
}
